```csharp
/// <summary>A point on the table plane, in inches. Origin at the table's corner; X along Width (0..W), Y along Length (0..L).</summary>
public readonly record struct Point2(double X, double Y)
{
    public static Point2 operator +(Point2 a, Point2 b) => new(a.X + b.X, a.Y + b.Y);
    public static Point2 operator -(Point2 a, Point2 b) => new(a.X - b.X, a.Y - b.Y);
    public static Point2 operator *(Point2 a, double k) => new(a.X * k, a.Y * k);
    public double Length => Math.Sqrt(X * X + Y * Y);
    public double DistanceTo(Point2 other) => (this - other).Length;
    public double Dot(Point2 other) => X * other.X + Y * other.Y;
    public double Cross(Point2 other) => X * other.Y - Y * other.X;

    /// <summary>Rotates about the origin. Yaw 0 faces +Y; positive yaw turns counter-clockwise seen from above.</summary>
    public Point2 Rotate(Yaw yaw)
    {
        var (s, c) = Math.SinCos(yaw.Degrees * Math.PI / 180);
        return new(X * c - Y * s, X * s + Y * c);
    }
}

/// <summary>A flat polygon on the table plane; deployment zones resolve to sets of these. Any vertex order; may be concave.</summary>
public sealed record Polygon(IReadOnlyList<Point2> Vertices)
{
    /// <summary>Even-odd point-in-polygon. A point on an edge or a vertex is inside.</summary>
    public bool Contains(Point2 p)
    {
        var v = Vertices;
        var inside = false;
        for (int i = 0, j = v.Count - 1; i < v.Count; j = i++)
        {
            var a = v[j];
            var b = v[i];
            if (OnSegment(a, b, p)) return true;
            if ((a.Y > p.Y) != (b.Y > p.Y))
            {
                var x = a.X + (p.Y - a.Y) * (b.X - a.X) / (b.Y - a.Y);
                if (p.X < x) inside = !inside;
            }
        }
        return inside;
    }

    internal static bool OnSegment(Point2 a, Point2 b, Point2 p)
    {
        var ab = b - a;
        var ap = p - a;
        if (Math.Abs(ab.Cross(ap)) > Tolerance.Distance * Math.Max(1, ab.Length)) return false;
        var t = ab.Dot(ap);
        return t >= -Tolerance.Distance && t <= ab.Dot(ab) + Tolerance.Distance;
    }
}
```

```csharp
namespace Tabletop.Geometry;

/// <summary>A closed convex outline on the table plane in world coordinates. Every base and every footprint becomes one of these.</summary>
internal abstract record Outline
{
    /// <summary>Boundary inclusive.</summary>
    public abstract bool Contains(Point2 p);
    /// <summary>Distance from p to the boundary, whether p is inside or outside.</summary>
    public abstract double BoundaryDistance(Point2 p);
    /// <summary>Nearest point of the outline to p; p itself when inside.</summary>
    public abstract Point2 ClosestPointTo(Point2 p);
    public double DistanceTo(Point2 p) => Contains(p) ? 0 : ClosestPointTo(p).DistanceTo(p);
    /// <summary>Does the segment a–b touch the outline anywhere.</summary>
    public abstract bool IntersectsSegment(Point2 a, Point2 b);
    /// <summary>Points on the boundary, for sampling sight lines.</summary>
    public abstract IReadOnlyList<Point2> BoundaryPoints(int count);

    /// <summary>The whole of inner lies inside this outline. Both convex.</summary>
    public bool ContainsOutline(Outline inner) => inner switch
    {
        CircleOutline c => Contains(c.Centre) && BoundaryDistance(c.Centre) >= c.Radius - Tolerance.Distance,
        PolygonOutline p => p.Vertices.All(Contains),
        _ => throw new ArgumentOutOfRangeException(nameof(inner)),
    };

    /// <summary>Shortest gap between two outlines and the points it is measured between. 0 when they touch or overlap.</summary>
    public static (double Distance, Point2 OnA, Point2 OnB) Separation(Outline a, Outline b)
    {
        switch (a, b)
        {
            case (CircleOutline ca, CircleOutline cb):
            {
                var d = cb.Centre - ca.Centre;
                var len = d.Length;
                var u = len < Tolerance.Distance ? new Point2(1, 0) : d * (1 / len);
                return (Math.Max(0, len - ca.Radius - cb.Radius), ca.Centre + u * ca.Radius, cb.Centre - u * cb.Radius);
            }
            case (CircleOutline c, PolygonOutline p):
            {
                var (dist, onP, onC) = CircleToPolygon(c, p);
                return (dist, onC, onP);
            }
            case (PolygonOutline p, CircleOutline c):
            {
                var (dist, onP, onC) = CircleToPolygon(c, p);
                return (dist, onP, onC);
            }
            case (PolygonOutline pa, PolygonOutline pb):
                return PolygonToPolygon(pa, pb);
            default:
                throw new ArgumentOutOfRangeException(nameof(a));
        }
    }

    static (double, Point2 OnPolygon, Point2 OnCircle) CircleToPolygon(CircleOutline c, PolygonOutline p)
    {
        if (p.Contains(c.Centre)) return (0, c.Centre, c.Centre);
        var q = p.ClosestBoundaryPoint(c.Centre);
        var d = q - c.Centre;
        var len = d.Length;
        return (Math.Max(0, len - c.Radius), q, c.Centre + d * (c.Radius / len));
    }

    static (double, Point2, Point2) PolygonToPolygon(PolygonOutline a, PolygonOutline b)
    {
        var best = (Distance: double.PositiveInfinity, OnA: a.Vertices[0], OnB: b.Vertices[0]);
        foreach (var v in a.Vertices)
        {
            var q = b.ClosestBoundaryPoint(v);
            var d = v.DistanceTo(q);
            if (d < best.Distance) best = (d, v, q);
        }
        foreach (var v in b.Vertices)
        {
            var q = a.ClosestBoundaryPoint(v);
            var d = v.DistanceTo(q);
            if (d < best.Distance) best = (d, q, v);
        }
        return InteriorsIntersect(a, b) ? (0, best.OnA, best.OnB) : best;
    }

    /// <summary>The two overlap with positive area. Touching edges do not count.</summary>
    public static bool InteriorsIntersect(Outline a, Outline b) => (a, b) switch
    {
        (CircleOutline ca, CircleOutline cb) => ca.Centre.DistanceTo(cb.Centre) < ca.Radius + cb.Radius - Tolerance.Distance,
        (CircleOutline c, PolygonOutline p) => p.DistanceTo(c.Centre) < c.Radius - Tolerance.Distance,
        (PolygonOutline p, CircleOutline c) => p.DistanceTo(c.Centre) < c.Radius - Tolerance.Distance,
        (PolygonOutline pa, PolygonOutline pb) => PolygonOutline.SeparatingAxisTest(pa, pb),
        _ => throw new ArgumentOutOfRangeException(nameof(a)),
    };

    internal static Point2 ClosestOnSegment(Point2 a, Point2 b, Point2 p)
    {
        var ab = b - a;
        var l2 = ab.Dot(ab);
        if (l2 < Tolerance.Distance * Tolerance.Distance) return a;
        return a + ab * Math.Clamp((p - a).Dot(ab) / l2, 0, 1);
    }

    internal static bool SegmentsIntersect(Point2 p1, Point2 p2, Point2 q1, Point2 q2)
    {
        var d1 = (q2 - q1).Cross(p1 - q1);
        var d2 = (q2 - q1).Cross(p2 - q1);
        var d3 = (p2 - p1).Cross(q1 - p1);
        var d4 = (p2 - p1).Cross(q2 - p1);
        if (((d1 > 0 && d2 < 0) || (d1 < 0 && d2 > 0)) && ((d3 > 0 && d4 < 0) || (d3 < 0 && d4 > 0))) return true;
        return Polygon.OnSegment(q1, q2, p1) || Polygon.OnSegment(q1, q2, p2) || Polygon.OnSegment(p1, p2, q1) || Polygon.OnSegment(p1, p2, q2);
    }
}

internal sealed record CircleOutline(Point2 Centre, double Radius) : Outline
{
    public override bool Contains(Point2 p) => Centre.DistanceTo(p) <= Radius + Tolerance.Distance;
    public override double BoundaryDistance(Point2 p) => Math.Abs(Centre.DistanceTo(p) - Radius);
    public override Point2 ClosestPointTo(Point2 p)
    {
        var d = p - Centre;
        var len = d.Length;
        return len <= Radius + Tolerance.Distance ? p : Centre + d * (Radius / len);
    }
    public override bool IntersectsSegment(Point2 a, Point2 b) => Centre.DistanceTo(ClosestOnSegment(a, b, Centre)) <= Radius + Tolerance.Distance;
    public override IReadOnlyList<Point2> BoundaryPoints(int count) => Enumerable.Range(0, count)
        .Select(i => { var t = 2 * Math.PI * i / count; return Centre + new Point2(Math.Cos(t), Math.Sin(t)) * Radius; })
        .ToList();
}

/// <summary>A convex polygon. The constructor orders the vertices counter-clockwise.</summary>
internal sealed record PolygonOutline : Outline
{
    public IReadOnlyList<Point2> Vertices { get; }

    public PolygonOutline(IReadOnlyList<Point2> vertices)
    {
        if (vertices.Count < 3) throw new ArgumentException("A polygon outline needs at least three vertices.", nameof(vertices));
        double twiceArea = 0;
        for (int i = 0, j = vertices.Count - 1; i < vertices.Count; j = i++) twiceArea += vertices[j].Cross(vertices[i]);
        Vertices = twiceArea < 0 ? vertices.Reverse().ToList() : vertices.ToList();
    }

    public override bool Contains(Point2 p)
    {
        for (int i = 0, j = Vertices.Count - 1; i < Vertices.Count; j = i++)
            if ((Vertices[i] - Vertices[j]).Cross(p - Vertices[j]) < -Tolerance.Distance) return false;
        return true;
    }

    public Point2 ClosestBoundaryPoint(Point2 p)
    {
        var best = Vertices[0];
        var bestD = double.PositiveInfinity;
        for (int i = 0, j = Vertices.Count - 1; i < Vertices.Count; j = i++)
        {
            var q = ClosestOnSegment(Vertices[j], Vertices[i], p);
            var d = q.DistanceTo(p);
            if (d < bestD) { bestD = d; best = q; }
        }
        return best;
    }

    public override double BoundaryDistance(Point2 p) => ClosestBoundaryPoint(p).DistanceTo(p);
    public override Point2 ClosestPointTo(Point2 p) => Contains(p) ? p : ClosestBoundaryPoint(p);

    public override bool IntersectsSegment(Point2 a, Point2 b)
    {
        if (Contains(a) || Contains(b)) return true;
        for (int i = 0, j = Vertices.Count - 1; i < Vertices.Count; j = i++)
            if (SegmentsIntersect(a, b, Vertices[j], Vertices[i])) return true;
        return false;
    }

    /// <summary>Every vertex and every edge midpoint; count is ignored (the polygon decides).</summary>
    public override IReadOnlyList<Point2> BoundaryPoints(int count)
    {
        var pts = new List<Point2>(Vertices.Count * 2);
        for (int i = 0, j = Vertices.Count - 1; i < Vertices.Count; j = i++)
        {
            pts.Add(Vertices[i]);
            pts.Add((Vertices[i] + Vertices[j]) * 0.5);
        }
        return pts;
    }

    internal static bool SeparatingAxisTest(PolygonOutline a, PolygonOutline b)
    {
        foreach (var axis in Axes(a).Concat(Axes(b)))
        {
            var (minA, maxA) = Project(a, axis);
            var (minB, maxB) = Project(b, axis);
            if (maxA <= minB + Tolerance.Distance || maxB <= minA + Tolerance.Distance) return false;
        }
        return true;
    }

    static IEnumerable<Point2> Axes(PolygonOutline p)
    {
        for (int i = 0, j = p.Vertices.Count - 1; i < p.Vertices.Count; j = i++)
        {
            var e = p.Vertices[i] - p.Vertices[j];
            yield return new Point2(-e.Y, e.X) * (1 / e.Length);
        }
    }

    static (double Min, double Max) Project(PolygonOutline p, Point2 axis)
    {
        double min = double.PositiveInfinity, max = double.NegativeInfinity;
        foreach (var v in p.Vertices) { var d = v.Dot(axis); min = Math.Min(min, d); max = Math.Max(max, d); }
        return (min, max);
    }
}
```

```csharp
namespace Tabletop.Geometry;

/// <summary>A model is a base and a height, upright, yaw only (geometry values → Height, position, facing).</summary>
public static class Bodies
{
    /// <summary>An oval base is a convex polygon with this many vertices; the error is under 0.2% of its radius.</summary>
    public const int OvalVertices = 64;

    /// <summary>Height of the base's underside: the settled Z.</summary>
    public static double Bottom(this PlacedModel m) => m.Placement.Position.Z;

    /// <summary>Top of the line-of-sight body.</summary>
    public static double Top(this PlacedModel m) => m.Placement.Position.Z + m.Body.Height;

    /// <summary>Where line of sight is drawn from: the top centre of the body.</summary>
    public static Point3 Eye(this PlacedModel m) => new(m.Placement.Position.X, m.Placement.Position.Y, m.Top());

    /// <summary>
    /// Rulebook p.11: other units are sight blockers. A body as a Solid, BlocksLineOfSight volume the caller can
    /// pass to <see cref="LineOfSight"/>. Oval and square bases become their bounding rectangle.
    /// </summary>
    public static WorldVolume AsBlockingVolume(Guid id, PlacedModel m)
    {
        var b = m.Body.Base;
        var footprint = b.Shape == BaseShape.Circle ? Footprint.Circle(b.A / 2) : Footprint.Rect(b.B, b.A);
        var volume = new Volume(VolumeKind.Solid, footprint, new Point2(0, 0), new Yaw(0), 0, m.Body.Height, TerrainDeclarations.BlocksLineOfSight, null);
        return new WorldVolume(id, volume, new TerrainTransform(m.Placement.Position, m.Placement.Yaw, 1));
    }

    internal static Outline BaseOutline(this PlacedModel m)
    {
        var centre = m.Placement.Position.Horizontal;
        var yaw = m.Placement.Yaw;
        var b = m.Body.Base;
        switch (b.Shape)
        {
            case BaseShape.Circle:
                return new CircleOutline(centre, b.A / 2);
            case BaseShape.Square:
            {
                var h = b.A / 2;
                return new PolygonOutline(Place([new(-h, -h), new(h, -h), new(h, h), new(-h, h)], centre, yaw));
            }
            case BaseShape.Oval:
            {
                var local = Enumerable.Range(0, OvalVertices).Select(i =>
                {
                    var t = 2 * Math.PI * i / OvalVertices;
                    return new Point2(b.B / 2 * Math.Cos(t), b.A / 2 * Math.Sin(t));
                });
                return new PolygonOutline(Place(local, centre, yaw));
            }
            default:
                throw new ArgumentOutOfRangeException(nameof(m), b.Shape, "Unknown base shape.");
        }
    }

    /// <summary>Local points → world: rotate by yaw about the origin, then translate to centre.</summary>
    internal static List<Point2> Place(IEnumerable<Point2> local, Point2 centre, Yaw yaw) => local.Select(p => p.Rotate(yaw) + centre).ToList();
}
```

```csharp
namespace Tabletop.Geometry;

/// <summary>A volume's prism in world coordinates: the piece's position, yaw and uniform scale applied (geometry values → The terrain transform).</summary>
public static class WorldVolumes
{
    public static double WorldBottom(this WorldVolume v) => v.Transform.Position.Z + v.Volume.Bottom * v.Transform.Scale;
    public static double WorldTop(this WorldVolume v) => v.Transform.Position.Z + v.Volume.Top * v.Transform.Scale;
    public static Point2 WorldCentre(this WorldVolume v) => v.Transform.Position.Horizontal + (v.Volume.Offset * v.Transform.Scale).Rotate(v.Transform.Yaw);
    public static Yaw WorldYaw(this WorldVolume v) => new(v.Transform.Yaw.Degrees + v.Volume.Yaw.Degrees);

    internal static Outline WorldOutline(this WorldVolume v)
    {
        var s = v.Transform.Scale;
        var centre = v.WorldCentre();
        var f = v.Volume.Footprint;
        if (f.Kind == FootprintKind.Circle) return new CircleOutline(centre, f.A * s);
        var hw = f.A * s / 2;
        var hl = f.B * s / 2;
        return new PolygonOutline(Bodies.Place([new(-hw, -hl), new(hw, -hl), new(hw, hl), new(-hw, hl)], centre, v.WorldYaw()));
    }

    internal static bool VerticalRangeContains(this WorldVolume v, double z) =>
        z >= v.WorldBottom() - Tolerance.Distance && z <= v.WorldTop() + Tolerance.Distance;

    internal static bool ContainsPoint(this WorldVolume v, Point3 p) => v.VerticalRangeContains(p.Z) && v.WorldOutline().Contains(p.Horizontal);

    /// <summary>Does the 3D segment pass through the prism: clip to the vertical range, then a 2D segment test.</summary>
    internal static bool IntersectsSegment(this WorldVolume v, Point3 a, Point3 b)
    {
        var bottom = v.WorldBottom();
        var top = v.WorldTop();
        double t0 = 0, t1 = 1;
        var dz = b.Z - a.Z;
        if (Math.Abs(dz) < Tolerance.Distance)
        {
            if (a.Z < bottom - Tolerance.Distance || a.Z > top + Tolerance.Distance) return false;
        }
        else
        {
            var tb = (bottom - a.Z) / dz;
            var tt = (top - a.Z) / dz;
            t0 = Math.Max(0, Math.Min(tb, tt));
            t1 = Math.Min(1, Math.Max(tb, tt));
            if (t0 > t1) return false;
        }
        var p = new Point2(a.X + (b.X - a.X) * t0, a.Y + (b.Y - a.Y) * t0);
        var q = new Point2(a.X + (b.X - a.X) * t1, a.Y + (b.Y - a.Y) * t1);
        return v.WorldOutline().IntersectsSegment(p, q);
    }
}
```

```csharp
namespace Tabletop.Geometry;

/// <summary>Edge-to-edge between bases, never centre-to-centre. Horizontal, in inches; 0 when the outlines touch or overlap.</summary>
public static class Distance
{
    public static double EdgeToEdge(PlacedModel a, PlacedModel b) => Outline.Separation(a.BaseOutline(), b.BaseOutline()).Distance;

    public static double EdgeToEdge(PlacedModel a, Point2 point) => a.BaseOutline().DistanceTo(point);

    /// <summary>Base edge to a placed volume's footprint edge.</summary>
    public static double EdgeToEdge(PlacedModel a, WorldVolume volume) => Outline.Separation(a.BaseOutline(), volume.WorldOutline()).Distance;

    /// <summary>Base edge to a marker's edge; a marker is a circle of a diameter (geometry values → Marker size).</summary>
    public static double EdgeToMarker(PlacedModel a, Point2 centre, double diameter) =>
        Outline.Separation(a.BaseOutline(), new CircleOutline(centre, diameter / 2)).Distance;

    /// <summary>The two edge points a measurement is drawn between (presence → A measurement is two points). Z is each base's settled height.</summary>
    public static (Point3 OnA, Point3 OnB) ClosestPoints(PlacedModel a, PlacedModel b)
    {
        var (_, pa, pb) = Outline.Separation(a.BaseOutline(), b.BaseOutline());
        return (new Point3(pa.X, pa.Y, a.Bottom()), new Point3(pb.X, pb.Y, b.Bottom()));
    }

    /// <summary>Edge-to-edge within range, inclusive.</summary>
    public static bool InRange(PlacedModel a, PlacedModel b, double range) => EdgeToEdge(a, b) <= range + Tolerance.Distance;

    /// <summary>Difference of the two base heights (rulebook p.9: "within 2" horizontally and 4" vertically").</summary>
    public static double Vertical(PlacedModel a, PlacedModel b) => Math.Abs(a.Bottom() - b.Bottom());
}
```

```csharp
namespace Tabletop.Geometry;

/// <summary>Raycast down: the highest Solid top under the point, or 0 (the slab) when nothing is.</summary>
public static class Settling
{
    public static double SurfaceHeightAt(Point2 point, IReadOnlyList<WorldVolume> volumes) =>
        SurfaceHeightAt(new Point3(point.X, point.Y, double.PositiveInfinity), volumes);

    /// <summary>Raycast down from <paramref name="from"/>: the highest Solid top at or below from.Z, or 0. Area volumes are passed through.</summary>
    public static double SurfaceHeightAt(Point3 from, IReadOnlyList<WorldVolume> volumes)
    {
        var best = 0.0;
        foreach (var v in volumes)
        {
            if (v.Volume.Kind != VolumeKind.Solid) continue;
            var top = v.WorldTop();
            if (top <= best || top > from.Z + Tolerance.Distance) continue;
            if (v.WorldOutline().Contains(from.Horizontal)) best = top;
        }
        return best;
    }
}
```

```csharp
namespace Tabletop.Geometry;

/// <summary>Which volumes a model is in. Computed, never stored (table domain model). Solid or Area alike; standing on a Solid's top counts.</summary>
public static class Containment
{
    /// <summary>Volumes whose footprint contains the base centre and whose vertical range contains the base height.</summary>
    public static IReadOnlyList<Guid> VolumesContaining(PlacedModel model, IReadOnlyList<WorldVolume> volumes)
    {
        var centre = model.Placement.Position.Horizontal;
        var z = model.Bottom();
        return volumes.Where(v => v.VerticalRangeContains(z) && v.WorldOutline().Contains(centre)).Select(v => v.Id).ToList();
    }

    /// <summary>Volumes whose footprint contains the whole base — rulebook p.11 "fully inside" — same vertical rule.</summary>
    public static IReadOnlyList<Guid> VolumesFullyContaining(PlacedModel model, IReadOnlyList<WorldVolume> volumes)
    {
        var outline = model.BaseOutline();
        var z = model.Bottom();
        return volumes.Where(v => v.VerticalRangeContains(z) && v.WorldOutline().ContainsOutline(outline)).Select(v => v.Id).ToList();
    }

    /// <summary>
    /// Rulebook p.14 and p.20: "must stay fully within 6" of it when exiting" — every point of the model's base within
    /// <paramref name="inches"/> of the container's base, measured edge to edge and horizontally, as every distance here is.
    /// Both outlines are convex, so the farthest point of the base is exact without sampling: a vertex for a polygon base,
    /// the centre pushed out by the radius for a circular one. When the bases overlap the circular form is an upper bound,
    /// which can only refuse a placement already touching the container.
    /// </summary>
    public static bool BaseFullyWithin(PlacedModel model, PlacedModel container, double inches)
    {
        var of = container.BaseOutline();
        var limit = inches + Tolerance.Distance;
        return model.BaseOutline() switch
        {
            CircleOutline c => of.DistanceTo(c.Centre) + c.Radius <= limit,
            PolygonOutline p => p.Vertices.All(v => of.DistanceTo(v) <= limit),
            var other => throw new ArgumentOutOfRangeException(nameof(model), other.GetType().Name),
        };
    }
}
```

```csharp
namespace Tabletop.Geometry;

/// <summary>Flagged, never prevented: base overlap and terrain intersection (architecture.md → Rules enforcement).</summary>
public static class Overlap
{
    /// <summary>Two bases share area and a vertical extent. Touching is not overlapping.</summary>
    public static bool Bases(PlacedModel a, PlacedModel b) =>
        IntervalsOverlap(a.Bottom(), a.Top(), b.Bottom(), b.Top()) && Outline.InteriorsIntersect(a.BaseOutline(), b.BaseOutline());

    /// <summary>The body is inside a Solid volume. Standing on its top is not; Area volumes never intersect.</summary>
    public static bool WithSolid(PlacedModel model, WorldVolume volume) =>
        volume.Volume.Kind == VolumeKind.Solid
        && IntervalsOverlap(model.Bottom(), model.Top(), volume.WorldBottom(), volume.WorldTop())
        && Outline.InteriorsIntersect(model.BaseOutline(), volume.WorldOutline());

    internal static bool IntervalsOverlap(double a0, double a1, double b0, double b1) => Math.Min(a1, b1) - Math.Max(a0, b0) > Tolerance.Distance;
}
```

```csharp
namespace Tabletop.Geometry;

/// <summary>Which volumes obstruct the line. Empty when clear.</summary>
public sealed record LineOfSightResult(bool Clear, IReadOnlyList<Guid> ObstructedBy);

/// <summary>
/// Line of sight between two base-and-height bodies against placed volumes. Only volumes declaring
/// BlocksLineOfSight obstruct; a SeeInAndOut volume obstructs a line passing through it but not one
/// that starts or ends inside it. The mesh is never consulted.
/// Lines run from the shooter's top centre to the target's top centre and to up to eight rim points at its
/// top and at its base — seventeen points for a circle or an oval, nine for a square; sight is clear when any
/// one line is unobstructed.
/// </summary>
public static class LineOfSight
{
    public const int RimSamples = 8;

    public static LineOfSightResult Check(PlacedModel from, PlacedModel to, IReadOnlyList<WorldVolume> volumes)
    {
        var eye = from.Eye();
        var blockers = Blockers(volumes);
        var seen = new HashSet<Guid>();
        var obstructed = new List<Guid>();
        foreach (var target in TargetPoints(to))
        {
            var hits = Obstructions(eye, target, blockers);
            if (hits.Count == 0) return new LineOfSightResult(true, []);
            foreach (var id in hits) if (seen.Add(id)) obstructed.Add(id);
        }
        return new LineOfSightResult(false, obstructed);
    }

    internal static List<WorldVolume> Blockers(IReadOnlyList<WorldVolume> volumes) =>
        volumes.Where(v => v.Volume.Declarations.HasFlag(TerrainDeclarations.BlocksLineOfSight)).ToList();

    /// <summary>The top centre, then the rim at the top, then the rim at the base.</summary>
    internal static IEnumerable<Point3> TargetPoints(PlacedModel to)
    {
        var p = to.Placement.Position;
        var rim = Rim(to.BaseOutline());
        yield return new Point3(p.X, p.Y, to.Top());
        foreach (var r in rim) yield return new Point3(r.X, r.Y, to.Top());
        foreach (var r in rim) yield return new Point3(r.X, r.Y, to.Bottom());
    }

    /// <summary>Eight points around the outline: evenly around a circle; every n-th vertex of a polygon.</summary>
    internal static IReadOnlyList<Point2> Rim(Outline outline) => outline switch
    {
        CircleOutline c => c.BoundaryPoints(RimSamples),
        PolygonOutline poly when poly.Vertices.Count <= RimSamples => poly.Vertices,
        PolygonOutline poly => Enumerable.Range(0, RimSamples).Select(i => poly.Vertices[i * poly.Vertices.Count / RimSamples]).ToList(),
        _ => throw new ArgumentOutOfRangeException(nameof(outline)),
    };

    /// <summary>Blocking volumes the segment passes through; a SeeInAndOut volume is skipped when either end is inside it.</summary>
    internal static List<Guid> Obstructions(Point3 p, Point3 q, IReadOnlyList<WorldVolume> blockers)
    {
        var hits = new List<Guid>();
        foreach (var v in blockers)
        {
            if (!v.IntersectsSegment(p, q)) continue;
            if (v.Volume.Declarations.HasFlag(TerrainDeclarations.SeeInAndOut) && (v.ContainsPoint(p) || v.ContainsPoint(q))) continue;
            hits.Add(v.Id);
        }
        return hits;
    }
```

```csharp
/// <summary>
/// The two geometric facts rulebook p.11's cover rule is built from. The majority rule and the modifier are the ruleset's.

public static class Cover
{
    /// <summary>Cover-declared volumes the whole base is inside: "fully inside a piece of cover terrain".</summary>
    public static IReadOnlyList<Guid> InsideCoverVolumes(PlacedModel model, IReadOnlyList<WorldVolume> volumes) =>
        Containment.VolumesFullyContaining(model, volumes.Where(v => v.Volume.Declarations.HasFlag(TerrainDeclarations.Cover)).ToList());

    /// <summary>
    /// Blocking volumes that stop at least one of the sight lines <see cref="LineOfSight.Check"/> draws, in input order.
    /// Non-empty while Check is clear means "behind a sight blocker".
    /// </summary>
    public static IReadOnlyList<Guid> SightBlockersBetween(PlacedModel from, PlacedModel to, IReadOnlyList<WorldVolume> volumes)
    {
        var eye = from.Eye();
        var blockers = LineOfSight.Blockers(volumes);
        var seen = new HashSet<Guid>();
        var ordered = new List<Guid>();
        foreach (var target in LineOfSight.TargetPoints(to))
            foreach (var id in LineOfSight.Obstructions(eye, target, blockers))
                if (seen.Add(id)) ordered.Add(id);
        return ordered;
    }
}
```

```csharp
namespace Tabletop.Geometry;

/// <summary>Which models are out of coherency, by index into the input. Flagged, never prevented.</summary>
public sealed record CoherencyResult(bool Coherent, IReadOnlyList<int> OutOfCoherency);

/// <summary>Rulebook p.7: within chainDistance of at least one other model, forming one uninterrupted chain, and within maxSpread of every other model in it.</summary>
public static class Coherency
{
    public static CoherencyResult Check(IReadOnlyList<PlacedModel> models, double chainDistance, double maxSpread)
    {
        var n = models.Count;
        if (n <= 1) return new CoherencyResult(true, []);

        var d = new double[n, n];
        for (var i = 0; i < n; i++)
            for (var j = i + 1; j < n; j++)
                d[i, j] = d[j, i] = Distance.EdgeToEdge(models[i], models[j]);

        var component = new int[n];
        Array.Fill(component, -1);
        var count = 0;
        for (var s = 0; s < n; s++)
        {
            if (component[s] >= 0) continue;
            var stack = new Stack<int>();
            stack.Push(s);
            component[s] = count;
            while (stack.Count > 0)
            {
                var i = stack.Pop();
                for (var j = 0; j < n; j++)
                    if (component[j] < 0 && d[i, j] <= chainDistance + Tolerance.Distance)
                    {
                        component[j] = count;
                        stack.Push(j);
                    }
            }
            count++;
        }

        var sizes = new int[count];
        foreach (var c in component) sizes[c]++;
        var main = Array.IndexOf(sizes, sizes.Max());
        var body = Enumerable.Range(0, n).Where(i => component[i] == main).ToList();

        var outOf = new List<int>();
        for (var i = 0; i < n; i++)
        {
            if (component[i] != main) outOf.Add(i);
            else if (body.Any(j => j != i && d[i, j] > maxSpread + Tolerance.Distance)) outOf.Add(i);
        }
        return new CoherencyResult(outOf.Count == 0, outOf);
    }
}
```

```csharp
{
    public static Polygon Bounds(TableSize size) =>

    public static bool ContainsBase(TableSize size, PlacedModel model) => Outline(size).ContainsOutline(model.BaseOutline());

    internal static PolygonOutline Outline(TableSize size) => new(Bounds(size).Vertices);
}
```

```csharp
namespace Tabletop.Geometry;

/// <summary>
/// The straight-line reach of a base centre from its start: one boundary vertex per ray, drawn by the client as a polygon.
/// A prediction for responsiveness; the server's event wins (architecture.md → Geometry).
/// </summary>
public sealed record MovementRegion(Point2 Origin, double Allowance, bool CappedByDifficultTerrain, IReadOnlyList<Point2> Boundary);

/// <summary>One straight move judged against the same parameters. Distance is the base's displacement (the ruler readout).</summary>
public sealed record MoveCheck(double Distance, double EffectiveAllowance, bool WithinAllowance, bool CrossesImpassable, bool EntersDifficult, bool LeavesTable);

/// <summary>
/// Renders the server's move parameters — allowance, the volumes that cannot be crossed, the difficult volumes and the cap
/// GameRules computed — as geometry (architecture.md → How the client stays responsive). Footprints are tested in 2D:
/// the server already chose which volumes count.
/// </summary>
public static class Movement
{
    public const int Rays = 180;
    public const double Step = 0.25;

    public static MovementRegion Region(PlacedModel start, double allowance, IReadOnlyList<WorldVolume> impassable,
        IReadOnlyList<WorldVolume> difficult, double? difficultCap, TableSize? table)
    {
        var origin = start.Placement.Position.Horizontal;
        var boundary = new List<Point2>(Rays);
        var capped = false;
        for (var r = 0; r < Rays; r++)
        {
            var angle = 2 * Math.PI * r / Rays;
            var dir = new Point2(Math.Cos(angle), Math.Sin(angle));
            var limit = allowance;
            var reached = 0.0;
            for (var k = 1; ; k++)
            {
                var d = Math.Min(k * Step, limit);
                var at = Displace(start, dir * d);
                if (Touches(at, impassable) || (table is { } t && !Table.ContainsBase(t, at))) break;
                if (difficultCap is { } cap && cap < limit && Touches(at, difficult))
                {
                    limit = cap;
                    capped = true;
                    if (d > limit + Tolerance.Distance) break;
                }
                reached = d;
                if (d >= limit - Tolerance.Distance) break;
            }
            boundary.Add(origin + dir * reached);
        }
        return new MovementRegion(origin, allowance, capped, boundary);
    }

    public static MoveCheck Check(PlacedModel start, Point2 destination, double allowance, IReadOnlyList<WorldVolume> impassable,
        IReadOnlyList<WorldVolume> difficult, double? difficultCap, TableSize? table)
    {
        var origin = start.Placement.Position.Horizontal;
        var delta = destination - origin;
        var distance = delta.Length;
        var steps = Math.Max(1, (int)Math.Ceiling(distance / Step));
        bool crosses = false, enters = false, leaves = false;
        for (var k = 0; k <= steps; k++)
        {
            var at = Displace(start, delta * ((double)k / steps));
            crosses |= Touches(at, impassable);
            enters |= Touches(at, difficult);
            leaves |= table is { } t && !Table.ContainsBase(t, at);
        }
        var effective = enters && difficultCap is { } cap ? Math.Min(allowance, cap) : allowance;
        return new MoveCheck(distance, effective, distance <= effective + Tolerance.Distance, crosses, enters, leaves);
    }

    /// <summary>
    /// Volumes whose footprint the base touches at any point of the straight path from start to destination, in input
    /// order. Rulebook p.11's "moves in or through" — a dangerous terrain test is owed for a piece crossed and left behind.
    /// </summary>
    public static IReadOnlyList<Guid> VolumesCrossed(PlacedModel start, Point2 destination, IReadOnlyList<WorldVolume> volumes)
    {
        var delta = destination - start.Placement.Position.Horizontal;
        var steps = Math.Max(1, (int)Math.Ceiling(delta.Length / Step));
        var touched = new HashSet<Guid>();
        for (var k = 0; k <= steps; k++)
        {
            var outline = Displace(start, delta * ((double)k / steps)).BaseOutline();
            foreach (var v in volumes)
                if (Outline.InteriorsIntersect(outline, v.WorldOutline())) touched.Add(v.Id);
        }
        return volumes.Where(v => touched.Contains(v.Id)).Select(v => v.Id).ToList();
    }

    static PlacedModel Displace(PlacedModel start, Point2 by)
    {
        var p = start.Placement.Position;
        return start with { Placement = start.Placement with { Position = new Point3(p.X + by.X, p.Y + by.Y, p.Z) } };
    }

    static bool Touches(PlacedModel at, IReadOnlyList<WorldVolume> volumes)
    {
        var outline = at.BaseOutline();
        return volumes.Any(v => Outline.InteriorsIntersect(outline, v.WorldOutline()));
    }
}
```

```csharp
namespace Tabletop.Geometry;

/// <summary>
/// The built-in deployment zone styles: Grimdark Future p.21 (the six basic styles) and p.49 (free-for-all corners).
/// A shape id is "<style>/<n>", n the team ordinal the setup phase assigns in order. A zone is a set of polygons.
/// Measurements are anchored to table features — edges, corners, edge midpoints — never scaled (geometry values → Deployment shapes).
/// </summary>
public static class ZoneStyles
{
    public const string Frontline = "frontline";
    public const string GroundWar = "ground-war";
    public const string SideBattle = "side-battle";
    public const string Disordered = "disordered";
    public const string Spearhead = "spearhead";
    public const string OpposingForces = "opposing-forces";
    public const string FreeForAll = "free-for-all";

    public static IReadOnlyList<string> All { get; } = [Frontline, GroundWar, SideBattle, Disordered, Spearhead, OpposingForces, FreeForAll];

    /// <summary>The style's shape ids in team-ordinal order — what SetZoneStyle carries. Two for the basic styles, four for free-for-all.</summary>
    public static IReadOnlyList<string> ShapesOf(string style)
    {
        var count = style == FreeForAll ? 4 : All.Contains(style) ? 2 : throw new ArgumentException($"Unknown zone style '{style}'.", nameof(style));
        return Enumerable.Range(1, count).Select(n => $"{style}/{n}").ToList();
    }

    /// <summary>The polygons a shape id resolves to on a table, in the table frame (origin at the table's corner, 0..W × 0..L).</summary>
    public static IReadOnlyList<Polygon> Resolve(string shapeId, TableSize table)
    {
        var slash = shapeId.LastIndexOf('/');
        if (slash < 0 || !int.TryParse(shapeId[(slash + 1)..], out var n))
            throw new ArgumentException($"Malformed shape id '{shapeId}'.", nameof(shapeId));
        var style = shapeId[..slash];
        var f = new Frame(table);
        var a = f.Long;      // the long edge, u from 0 to a
        var b = f.Short;     // the short edge, v from 0 to b
        IReadOnlyList<IReadOnlyList<Point2>> shapes = (style, n) switch
        {
            // Frontline (also the core rules' deployment, p.6): 12" in from each long edge, the full width.
            (Frontline, 1) => [Rect(0, a, b - 12, b)],
            (Frontline, 2) => [Rect(0, a, 0, 12)],
            // Ground War: 24" in from each short edge.
            (GroundWar, 1) => [Rect(0, 24, 0, b)],
            (GroundWar, 2) => [Rect(a - 24, a, 0, b)],
            // Side Battle: a triangle in each of two opposite corners; its hypotenuse ends 18" short of the far corners along both edges (54" and 30" legs on 72×48, a 24" band between).
            (SideBattle, 1) => [Tri(P(0, 0), P(Leg(a), 0), P(0, Leg(b)))],
            (SideBattle, 2) => [Tri(P(a, b), P(a - Leg(a), b), P(a, b - Leg(b)))],
            // Disordered: a diamond through the edge midpoints; one team has both corners on one long edge, the other team the other two.
            (Disordered, 1) => [Tri(P(0, b / 2), P(a / 2, b), P(0, b)), Tri(P(a / 2, b), P(a, b / 2), P(a, b))],
            (Disordered, 2) => [Tri(P(0, 0), P(a / 2, 0), P(0, b / 2)), Tri(P(a / 2, 0), P(a, 0), P(a, b / 2))],
            // Spearhead: a triangle on each short edge, apex 24" in at the edge's centre.
            (Spearhead, 1) => [Tri(P(0, 0), P(24, b / 2), P(0, b))],
            (Spearhead, 2) => [Tri(P(a, b), P(a - 24, b / 2), P(a, 0))],
            // Opposing Forces: a 24"×24" square in two diagonally opposite corners.
            (OpposingForces, 1) => [Rect(0, 24, b - 24, b)],
            (OpposingForces, 2) => [Rect(a - 24, a, 0, 24)],
            // Free-for-all (p.49): a 2'×1' area in each corner — 24" along the long edge, 12" along the short — clockwise from the far-left corner (u = 0, v = b).
            (FreeForAll, 1) => [Rect(0, 24, b - 12, b)],
            (FreeForAll, 2) => [Rect(a - 24, a, b - 12, b)],
            (FreeForAll, 3) => [Rect(a - 24, a, 0, 12)],
            (FreeForAll, 4) => [Rect(0, 24, 0, 12)],
            _ => throw new ArgumentException($"Unknown shape id '{shapeId}'.", nameof(shapeId)),
        };
        return shapes.Select(s => new Polygon(s.Select(f.ToTable).ToList())).ToList();
    }

    public static bool Contains(string shapeId, TableSize table, Point2 p) => Resolve(shapeId, table).Any(z => z.Contains(p));

    /// <summary>Rulebook p.6 "fully within": the whole base inside one of the zone's polygons.</summary>
    public static bool BaseFullyWithin(string shapeId, TableSize table, PlacedModel model)
    {
        var outline = model.BaseOutline();
        return Resolve(shapeId, table).Any(z => new PolygonOutline(z.Vertices).ContainsOutline(outline));
    }

    static double Leg(double edge) => Math.Max(0, edge - 18);
    static Point2 P(double u, double v) => new(u, v);
    static IReadOnlyList<Point2> Tri(Point2 p, Point2 q, Point2 r) => [p, q, r];
    static IReadOnlyList<Point2> Rect(double u0, double u1, double v0, double v1) => [P(u0, v0), P(u1, v0), P(u1, v1), P(u0, v1)];

    /// <summary>Canonical frame: u along the long edge from 0, v along the short edge from 0, the origin the table's corner. A square table takes X as long.</summary>
    readonly record struct Frame(TableSize Table)
    {
        public bool XIsLong => Table.Width >= Table.Length;
        public double Long => XIsLong ? Table.Width : Table.Length;
        public double Short => XIsLong ? Table.Length : Table.Width;
        public Point2 ToTable(Point2 uv) => XIsLong ? uv : new(uv.Y, uv.X);
    }
}
```

## Questions for the project owner

1. **Where is the origin? Closed 2026-09-06.** This plan first assumed the table's centre; the compatibility review of 2026-09-05 **ruled the corner** (`0 ≤ x ≤ width`, `0 ≤ y ≤ length`, Z up), because the game server plan's `DraftEditor.OnSlab` and the contracts plan's samples already use it, nothing in the specs prefers the centre, and one frame on the wire and in the library removes the client shell's conversion layer. The owner ratified the corner on 2026-09-06, which settles a convention the wire, this library, both servers and the client all share. `Table`, `ZoneStyles`, the `Movement` tests, Global Constraints and spec correction 4 already match it, so no mechanism changes.
2. **What is the sight-line policy?** Assumed: eye at the shooter's top centre, the target's top centre plus up to eight rim points at its top and its base (seventeen for a circle or an oval, nine for a square), clear if any line is unobstructed (Task 8). A single centre-to-centre line is simpler and stricter, but then a low wall either hides a model wholly or not at all, and "behind a sight blocker" (`Cover.SightBlockersBetween`) has nothing to measure. A denser sample or a shooter-rim sample changes only `TargetPoints`/`Eye`.
3. **What does "mostly inside" mean for a single model?** Assumed: base centre inside the volume (`Containment.VolumesContaining`). If it must be area — more than half the base's area inside — the plan gains polygon clipping in `Outline` and a `BaseCoverage(PlacedModel, WorldVolume) : double`.
4. **Is straight-line reach enough for the movement region?** Assumed yes: `Movement.Region` is star-shaped about the start, since it is a cosmetic prediction and the server judges the landed move. Geodesic reach around corners (a visibility graph over footprint polygons) is a separate, larger task; the alternative in the other direction is a plain disc, which drops Tasks 12's obstacle handling.
5. **Free-for-all's "2'x1' area" — which way round?** Assumed 24" along the long edge by 12" along the short (Task 13). Swapping is a two-number change per corner.
6. **Side Battle's anchoring.** The diagram gives 54" and 30" legs with a 24" band on 72×48; assumed the anchored reading "the hypotenuse ends 18" short of the far corner along each edge" (`Leg(edge) = edge − 18`), which reproduces the diagram and shrinks sensibly. Fixed 54"/30" legs would overrun a 30×22 skirmish table; proportional legs are the normalisation the spec rejects. Note the diagram's three labels are not exactly consistent — 54" and 30" legs put the perpendicular gap between the hypotenuses at 24.5", not the 24" the arrow says — so the legs are taken as the shape and the band as illustration.
7. **Ovals as 64-gons?** Assumed acceptable (error under 0.2% of the radius, well inside the tolerance the architecture allows). Exact ellipse–ellipse distance has no closed form and would need an iterative solver on both client and server.
8. **Is a terrain container needed as well as a model container?** [Game flow](../specs/2026-08-28-game-flow-design.md) → Inside a container says a container is *"a model or a piece of terrain"*. `BaseFullyWithin` takes a `PlacedModel` container only, because only `Transport(X)` is automated (the interpreter plan's Task 19); a solid building holds models through the client's right-click menu, and the menu places by hand. If a building must offer the same 6" exit region, this gains one overload — `bool BaseFullyWithin(PlacedModel model, WorldVolume container, double inches)` over `WorldOutline()` instead of the container's `BaseOutline()`, five lines and one test — and `MoveRegion.Containment.ExitingId` then names a `TerrainVolume` id rather than a model id, which the record already permits since it is a bare `Guid`.
